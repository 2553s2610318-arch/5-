
import streamlit as st
import random

st.set_page_config(
    page_title="반장 마음 위로소",
    page_icon="🌷",
    layout="centered"
)

# 데이터
emotion_messages = {
    "속상해요": """
🌷 속상한 마음이 드는 건 당연해요.

반장은 친구들을 위해 여러 번 이야기하고 노력했을 거예요.
친구들이 바로 따라주지 않는다고 해서 반장이 잘못한 것은 아니에요.

오늘도 책임감을 가지고 노력한 당신은 충분히 멋진 반장이에요.
""",
    "화가 나요": """
🌷 화가 나는 마음도 자연스러운 감정이에요.

열심히 말했는데 들어주지 않으면 누구라도 속상하고 화가 날 수 있어요.

하지만 지금까지 노력해온 모습은 분명 가치 있는 일이에요.
""",
    "지쳤어요": """
🌷 많이 지쳤군요.

반장은 생각보다 많은 책임을 맡고 있어요.
친구들을 챙기고, 선생님 말씀도 전달하고, 분위기도 신경 써야 하니까요.

잠시 쉬어도 괜찮아요.
""",
    "혼란스러워요": """
🌷 어떻게 해야 할지 고민되는 순간이 있죠.

모든 친구를 한 번에 움직이게 만드는 것은 어른에게도 어려운 일이에요.

혼란스러운 마음이 드는 것은 최선을 다하고 있다는 증거일 수 있어요.
"""
}

encouragements = [
    "🌟 당신의 노력은 분명 누군가에게 도움이 되고 있어요.",
    "🌟 좋은 반장은 완벽한 사람이 아니라 계속 노력하는 사람이에요.",
    "🌟 오늘도 반 역할을 해낸 것만으로도 대단해요.",
    "🌟 친구들이 당장은 몰라도 당신의 진심은 가치가 있어요.",
    "🌟 책임감을 가지고 행동한 당신을 응원해요.",
    "🌟 반장을 맡아준 것 자체가 고마운 일이에요."
]

compliments = [
    "💖 책임감이 강한 반장이에요.",
    "💖 친구들을 생각하는 따뜻한 마음이 있어요.",
    "💖 포기하지 않고 노력하는 모습이 멋져요.",
    "💖 반을 위해 애쓰는 마음이 훌륭해요.",
    "💖 용기 있게 역할을 맡은 사람이에요.",
    "💖 다른 사람을 배려할 줄 아는 사람이에요."
]

st.title("🌷 반장 마음 위로소")
st.caption("속상한 반장을 위한 작은 응원 공간")

st.divider()

# 감정 선택
st.subheader("😊 지금 어떤 기분인가요?")

emotion = st.selectbox(
    "현재 마음을 선택하세요",
    list(emotion_messages.keys())
)

st.info(emotion_messages[emotion])

st.divider()

# 이야기 적기
st.subheader("✍️ 오늘 있었던 일을 적어보세요")

story = st.text_area(
    "무슨 일이 있었나요?",
    height=150,
    placeholder="예) 친구들에게 정리하자고 여러 번 말했는데 잘 듣지 않았어요."
)

if st.button("💌 위로 받기"):
    if story.strip():
        st.success(
            f"""
당신의 이야기를 읽었어요.

'{story[:60]}'

친구들이 바로 따라주지 않아 속상했겠어요.
그만큼 반을 생각하고 책임감을 느끼고 있다는 뜻이기도 해요.

오늘 있었던 일 때문에 스스로를 나쁘게 생각하지 않았으면 좋겠어요.
당신은 이미 충분히 애쓰고 있는 멋진 반장이에요. 🌷
"""
        )
    else:
        st.warning("이야기를 먼저 적어주세요.")

st.divider()

# 응원 버튼
st.subheader("🎁 반장 응원 받기")

if st.button("🌟 응원 메시지 보기"):
    st.success(random.choice(encouragements))

st.divider()

# 칭찬 카드
st.subheader("🏆 오늘의 반장 칭찬 카드")

if st.button("💖 칭찬 카드 뽑기"):
    st.balloons()
    st.success(random.choice(compliments))

st.divider()

st.markdown(
    """
### 🌼 기억해 주세요

반장은 모든 친구를 완벽하게 움직이는 사람이 아니에요.

반장은 친구들을 위해 노력하고,
필요한 말을 용기 있게 전하는 사람이에요.

오늘도 정말 수고했어요. 🌷
"""
)
